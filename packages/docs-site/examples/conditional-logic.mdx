This example demonstrates building workflows with conditional logic and branching paths.

<Warning>
  **Work in Progress:** Control flow nodes (If/Else, Switch) are planned for a future release. This example shows the conceptual approach.
</Warning>

## Overview

Conditional workflows allow:
1. Branch execution based on conditions
2. Skip nodes when conditions aren't met
3. Dynamic path selection
4. Conditional data transformation

## Planned Node Types

### IfNode (Planned)

```typescript
import { Node, defineNode, Input, Output, Property } from '@reactflo/core';

@defineNode({
  type: 'control.if',
  label: 'If Condition',
  category: 'Control Flow'
})
export class IfNode extends Node {
  @Input({ type: 'any', label: 'Value' })
  value: any;

  @Property({
    type: 'select',
    label: 'Operator',
    defaultValue: 'equals',
    options: [
      { value: 'equals', label: 'Equals' },
      { value: 'notEquals', label: 'Not Equals' },
      { value: 'greater', label: 'Greater Than' },
      { value: 'less', label: 'Less Than' },
      { value: 'truthy', label: 'Is Truthy' },
      { value: 'falsy', label: 'Is Falsy' }
    ]
  })
  operator: string = 'equals';

  @Property({
    type: 'string',
    label: 'Compare Value',
    defaultValue: ''
  })
  compareValue: string = '';

  @Output({ type: 'boolean', label: 'Condition' })
  condition: boolean;

  @Output({ type: 'any', label: 'Then' })
  thenOutput: any;

  @Output({ type: 'any', label: 'Else' })
  elseOutput: any;

  execute() {
    let result = false;

    switch (this.operator) {
      case 'equals':
        result = this.value == this.compareValue;
        break;
      case 'notEquals':
        result = this.value != this.compareValue;
        break;
      case 'greater':
        result = this.value > Number(this.compareValue);
        break;
      case 'less':
        result = this.value < Number(this.compareValue);
        break;
      case 'truthy':
        result = !!this.value;
        break;
      case 'falsy':
        result = !this.value;
        break;
    }

    this.condition = result;
    
    if (result) {
      this.thenOutput = this.value;
      this.elseOutput = undefined;
    } else {
      this.thenOutput = undefined;
      this.elseOutput = this.value;
    }
  }
}
```

### SwitchNode (Planned)

```typescript
@defineNode({
  type: 'control.switch',
  label: 'Switch',
  category: 'Control Flow'
})
export class SwitchNode extends Node {
  @Input({ type: 'any', label: 'Value' })
  value: any;

  @Property({
    type: 'string',
    label: 'Cases (comma-separated)',
    defaultValue: 'case1,case2,case3'
  })
  cases: string = 'case1,case2,case3';

  @Output({ type: 'any', label: 'Case 1' })
  case1: any;

  @Output({ type: 'any', label: 'Case 2' })
  case2: any;

  @Output({ type: 'any', label: 'Case 3' })
  case3: any;

  @Output({ type: 'any', label: 'Default' })
  default: any;

  execute() {
    const caseList = this.cases.split(',').map(c => c.trim());
    
    const matchIndex = caseList.indexOf(String(this.value));
    
    if (matchIndex >= 0) {
      // Match found
      this[`case${matchIndex + 1}`] = this.value;
    } else {
      // No match - use default
      this.default = this.value;
    }
  }
}
```

## Current Workaround: Filter-Based Branching

While control flow nodes are in development, you can implement basic conditional logic using data filtering:

### ConditionalFilterNode

```typescript
@defineNode({
  type: 'data.conditional-filter',
  label: 'Conditional Filter',
  category: 'Data'
})
export class ConditionalFilterNode extends Node {
  @Input({ type: 'any[]', label: 'Items' })
  items: any[] = [];

  @Property({
    type: 'select',
    label: 'Condition',
    defaultValue: 'greater',
    options: [
      { value: 'greater', label: 'Greater Than' },
      { value: 'less', label: 'Less Than' },
      { value: 'equals', label: 'Equals' }
    ]
  })
  condition: string = 'greater';

  @Property({
    type: 'number',
    label: 'Threshold',
    defaultValue: 10
  })
  threshold: number = 10;

  @Output({ type: 'any[]', label: 'Passed' })
  passed: any[];

  @Output({ type: 'any[]', label: 'Failed' })
  failed: any[];

  execute() {
    this.passed = [];
    this.failed = [];

    for (const item of this.items) {
      let condition = false;

      switch (this.condition) {
        case 'greater':
          condition = item > this.threshold;
          break;
        case 'less':
          condition = item < this.threshold;
          break;
        case 'equals':
          condition = item === this.threshold;
          break;
      }

      if (condition) {
        this.passed.push(item);
      } else {
        this.failed.push(item);
      }
    }
  }
}
```

### Example Usage

```typescript
import { Workflow } from '@reactflo/core';
import {
  DataSourceNode,
  ConditionalFilterNode,
  AggregateNode,
  DisplayNode
} from './nodes';

const workflow = new Workflow('conditional-workflow', 'Conditional Processing');

// Data source
const source = workflow.addNode(DataSourceNode, {
  position: { x: 100, y: 200 },
  data: { dataJson: '[5, 15, 8, 20, 3, 12]' }
});

// Conditional filter (split at 10)
const conditionalFilter = workflow.addNode(ConditionalFilterNode, {
  position: { x: 350, y: 200 },
  data: { condition: 'greater', threshold: 10 }
});

// Aggregate passed items
const aggregateHigh = workflow.addNode(AggregateNode, {
  position: { x: 600, y: 100 },
  data: { aggregateFunc: 'sum' }
});

// Aggregate failed items
const aggregateLow = workflow.addNode(AggregateNode, {
  position: { x: 600, y: 300 },
  data: { aggregateFunc: 'sum' }
});

// Display results
const displayHigh = workflow.addNode(DisplayNode, {
  position: { x: 850, y: 100 }
});

const displayLow = workflow.addNode(DisplayNode, {
  position: { x: 850, y: 300 }
});

// Connect nodes
workflow.connect(source.id, 'items', conditionalFilter.id, 'items');
workflow.connect(conditionalFilter.id, 'passed', aggregateHigh.id, 'items');
workflow.connect(conditionalFilter.id, 'failed', aggregateLow.id, 'items');
workflow.connect(aggregateHigh.id, 'result', displayHigh.id, 'value');
workflow.connect(aggregateLow.id, 'result', displayLow.id, 'value');

// Execute
await workflow.execute();
// Passed (>10): [15, 20, 12] -> Sum: 47
// Failed (≤10): [5, 8, 3] -> Sum: 16
```

## React Component

```tsx
import React from 'react';
import { WorkflowBuilder } from '@reactflo/react';
import {
  NumberInputNode,
  ConditionalFilterNode,
  DataSourceNode,
  AggregateNode,
  DisplayNode
} from './nodes';

function ConditionalWorkflow() {
  const nodes = [
    NumberInputNode,
    ConditionalFilterNode,
    DataSourceNode,
    AggregateNode,
    DisplayNode
  ];

  const handleExecute = (result) => {
    console.log('Conditional workflow executed:', result);
  };

  return (
    <div className="conditional-workflow">
      <h1>Conditional Logic Workflow</h1>
      <p className="notice">
        ⚠️ Full control flow nodes (If/Else, Switch) coming in future release
      </p>
      
      <WorkflowBuilder
        nodes={nodes}
        onExecute={handleExecute}
        showNodePalette={true}
        showPropertyPanel={true}
        style={{ height: '600px' }}
      />
    </div>
  );
}

export default ConditionalWorkflow;
```
